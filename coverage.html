package config

import (

// Config represents the main configuration structure for the copyright tool.
type Config struct {
        Copyright CopyrightConfig `yaml:"copyright" mapstructure:"copyright"`
        License   LicenseConfig   `yaml:"license" mapstructure:"license"`
        Scan      ScanConfig      `yaml:"scan" mapstructure:"scan"`
        Detection DetectionConfig `yaml:"detection" mapstructure:"detection"`
        Output    OutputConfig    `yaml:"output" mapstructure:"output"`
}

// CopyrightConfig contains copyright header configuration.
type CopyrightConfig struct {
        // Template is the copyright header template.
        // Supports placeholders: {{.Year}}, {{.Company}}, {{.Author}}, {{.SPDX}}
        Template string `yaml:"template" mapstructure:"template"`

        // Year is the copyright year. Use "auto" for current year.
        Year string `yaml:"year" mapstructure:"year"`

        // Company is the company or organization name.
        Company string `yaml:"company" mapstructure:"company"`

        // Author is the author name (optional).
        Author string `yaml:"author" mapstructure:"author"`

        // SPDXIdentifier is the SPDX license identifier (e.g., "Apache-2.0", "MIT").
        SPDXIdentifier string `yaml:"spdx_identifier" mapstructure:"spdx_identifier"`
}

// LicenseConfig contains license-related configuration.
type LicenseConfig struct {
        // Type is the license type (e.g., "apache-2.0", "mit", "custom").
        Type string `yaml:"type" mapstructure:"type"`

        // CustomText is the custom license text (used when Type is "custom").
        CustomText string `yaml:"custom_text" mapstructure:"custom_text"`

        // AutoCreate indicates whether to auto-create LICENSE file if missing.
        AutoCreate bool `yaml:"auto_create" mapstructure:"auto_create"`
}

// ScanConfig contains file scanning configuration.
type ScanConfig struct {
        // ExcludePatterns is a list of glob patterns for files/directories to exclude.
        ExcludePatterns []string `yaml:"exclude_patterns" mapstructure:"exclude_patterns"`

        // IncludePatterns is a list of glob patterns for files to include.
        IncludePatterns []string `yaml:"include_patterns" mapstructure:"include_patterns"`

        // SkipHidden indicates whether to skip hidden files and directories.
        SkipHidden bool `yaml:"skip_hidden" mapstructure:"skip_hidden"`

        // FollowSymlinks indicates whether to follow symbolic links.
        FollowSymlinks bool `yaml:"follow_symlinks" mapstructure:"follow_symlinks"`

        // ExcludedExtensions is a list of file extensions to exclude.
        ExcludedExtensions []string `yaml:"excluded_extensions" mapstructure:"excluded_extensions"`
}

// DetectionConfig contains copyright detection configuration.
type DetectionConfig struct {
        // LinesToScan is the number of lines from the start of file to scan.
        LinesToScan int `yaml:"lines_to_scan" mapstructure:"lines_to_scan"`

        // Keywords is a list of keywords that indicate copyright presence.
        Keywords []string `yaml:"keywords" mapstructure:"keywords"`

        // RequireAll indicates whether all keywords must be present.
        RequireAll bool `yaml:"require_all" mapstructure:"require_all"`
}

// OutputConfig contains output formatting configuration.
type OutputConfig struct {
        // Format is the output format (e.g., "text", "json", "junit").
        Format string `yaml:"format" mapstructure:"format"`

        // Verbose enables verbose output.
        Verbose bool `yaml:"verbose" mapstructure:"verbose"`

        // Quiet suppresses non-error output.
        Quiet bool `yaml:"quiet" mapstructure:"quiet"`
}

// Load reads and parses the configuration file using Viper.
// It looks for .copyright.yaml in the current directory and applies defaults.
func Load(configPath string) (*Config, error) {
        v := viper.New()

        // Set defaults
        setDefaults(v)

        // Configure Viper
        v.SetConfigName(".copyright")
        v.SetConfigType("yaml")

        if configPath != "" {
                // Use specified config file
                v.SetConfigFile(configPath)
        } else {
                // Search for config in current directory
                v.AddConfigPath(".")
                v.AddConfigPath("$HOME")
        }

        // Read config file
        if err := v.ReadInConfig(); err != nil {
                // Config file not found is acceptable - we'll use defaults
                if _, ok := err.(viper.ConfigFileNotFoundError); !ok {
                        return nil, fmt.Errorf("failed to read config file: %w", err)
                }
        }

        // Unmarshal config
        var cfg Config
        if err := v.Unmarshal(&cfg); err != nil {
                return nil, fmt.Errorf("failed to unmarshal config: %w", err)
        }

        // Validate configuration
        if err := cfg.Validate(); err != nil {
                return nil, fmt.Errorf("invalid configuration: %w", err)
        }

        return &cfg, nil
}

// LoadOrDefault loads the configuration or returns defaults if config file is not found.
func LoadOrDefault(configPath string) *Config {
        cfg, err := Load(configPath)
        if err != nil {
                // If config fails to load, return defaults
                return NewDefault()
        }
        return cfg
}

// Validate performs validation on the configuration.
func (c *Config) Validate() error {
        if err := c.validateCopyright(); err != nil {
                return fmt.Errorf("copyright config: %w", err)
        }

        if err := c.validateLicense(); err != nil {
                return fmt.Errorf("license config: %w", err)
        }

        if err := c.validateScan(); err != nil {
                return fmt.Errorf("scan config: %w", err)
        }

        if err := c.validateDetection(); err != nil {
                return fmt.Errorf("detection config: %w", err)
        }

        if err := c.validateOutput(); err != nil {
                return fmt.Errorf("output config: %w", err)
        }

        return nil
}

// validateCopyright validates the copyright configuration.
func (c *Config) validateCopyright() error {
        if c.Copyright.Template == "" {
                return fmt.Errorf("template cannot be empty")
        }

        // Validate year format
        if c.Copyright.Year != "" && c.Copyright.Year != "auto" {
                // Check if it's a valid year format (either single year or range)
                if !isValidYearFormat(c.Copyright.Year) {
                        return fmt.Errorf("invalid year format: %s (expected format: YYYY, YYYY-YYYY, or 'auto')", c.Copyright.Year)
                }
        }

        return nil
}

// validateLicense validates the license configuration.
func (c *Config) validateLicense() error {
        if c.License.Type == "" {
                // License type is optional
                return nil
        }

        // Normalize license type to lowercase
        licenseType := strings.ToLower(c.License.Type)

        // Check if it's a valid license type
        if !isValidLicenseType(licenseType) {
                return fmt.Errorf("invalid license type: %s (valid types: %s)", c.License.Type, strings.Join(ValidLicenseTypes, ", "))
        }

        // If custom license, ensure custom text is provided
        if licenseType == "custom" && c.License.CustomText == "" {
                return fmt.Errorf("custom_text must be provided when license type is 'custom'")
        }

        return nil
}

// validateScan validates the scan configuration.
func (c *Config) validateScan() error {
        // Validate exclude patterns
        for _, pattern := range c.Scan.ExcludePatterns {
                if pattern == "" {
                        return fmt.Errorf("exclude_patterns cannot contain empty strings")
                }
        }

        // Validate include patterns
        for _, pattern := range c.Scan.IncludePatterns {
                if pattern == "" {
                        return fmt.Errorf("include_patterns cannot contain empty strings")
                }
        }

        // Validate excluded extensions format
        for _, ext := range c.Scan.ExcludedExtensions {
                if ext == "" {
                        return fmt.Errorf("excluded_extensions cannot contain empty strings")
                }
                if !strings.HasPrefix(ext, ".") {
                        return fmt.Errorf("excluded extension must start with '.': %s", ext)
                }
        }

        return nil
}

// validateDetection validates the detection configuration.
func (c *Config) validateDetection() error {
        if c.Detection.LinesToScan <= 0 {
                return fmt.Errorf("lines_to_scan must be positive, got: %d", c.Detection.LinesToScan)
        }

        if c.Detection.LinesToScan > 1000 {
                return fmt.Errorf("lines_to_scan is too large (max: 1000), got: %d", c.Detection.LinesToScan)
        }

        if len(c.Detection.Keywords) == 0 {
                return fmt.Errorf("keywords list cannot be empty")
        }

        // Validate keywords are not empty
        for i, keyword := range c.Detection.Keywords {
                if strings.TrimSpace(keyword) == "" {
                        return fmt.Errorf("keyword at index %d cannot be empty or whitespace", i)
                }
        }

        return nil
}

// validateOutput validates the output configuration.
func (c *Config) validateOutput() error {
        if c.Output.Format == "" {
                return fmt.Errorf("output format cannot be empty")
        }

        // Validate output format
        if !isValidOutputFormat(c.Output.Format) {
                return fmt.Errorf("invalid output format: %s (valid formats: %s)", c.Output.Format, strings.Join(ValidOutputFormats, ", "))
        }

        // Verbose and quiet are mutually exclusive
        if c.Output.Verbose && c.Output.Quiet {
                return fmt.Errorf("verbose and quiet options are mutually exclusive")
        }

        return nil
}

// isValidYearFormat checks if the year string is in a valid format.
func isValidYearFormat(year string) bool {
        // Allow "auto"
        if year == "auto" {
                return true
        }

        // Allow single year (YYYY)
        if len(year) == 4 {
                for _, c := range year {
                        if c < '0' || c > '9' {
                                return false
                        }
                }
                return true
        }

        // Allow year range (YYYY-YYYY)
        if len(year) == 9 && year[4] == '-' {
                startYear := year[0:4]
                endYear := year[5:9]

                for _, c := range startYear {
                        if c < '0' || c > '9' {
                                return false
                        }
                }
                for _, c := range endYear {
                        if c < '0' || c > '9' {
                                return false
                        }
                }
                return true
        }

        return false
}

// isValidLicenseType checks if the license type is valid.
func isValidLicenseType(licenseType string) bool {
        normalized := strings.ToLower(licenseType)
        for _, valid := range ValidLicenseTypes {
                if normalized == valid {
                        return true
                }
        }
        return false
}

// isValidOutputFormat checks if the output format is valid.
func isValidOutputFormat(format string) bool {
        normalized := strings.ToLower(format)
        for _, valid := range ValidOutputFormats {
                if normalized == valid {
                        return true
                }
        }
        return false
}

// CheckResult represents the result of checking a single file for copyright notices.
type CheckResult struct {
        FilePath         string   // Absolute path to the checked file
        HasCopyright     bool     // True if copyright was detected
        DetectedKeywords []string // List of detected keywords
        LineFound        int      // 1-indexed line number where first keyword was found, 0 if not found
        Error            error    // Error encountered during checking, if any
}

// CheckerOptions configures the copyright checker behavior.
type CheckerOptions struct {
        LinesToScan int      // Number of lines from file start to scan (default: 20)
        Keywords    []string // Keywords to detect, case-insensitive (default: DefaultKeywords)
        RequireAll  bool     // If true, all keywords must be present; if false, any keyword matches
}

// DefaultCheckerOptions returns checker options with sensible defaults.
func DefaultCheckerOptions() *CheckerOptions {
        return &CheckerOptions{
                LinesToScan: 20,
                Keywords:    DefaultKeywords,
                RequireAll:  false,
        }
}

// Checker detects copyright notices in files.
type Checker struct {
        options *CheckerOptions
}

// NewChecker creates a new copyright checker with the given options.
// If options is nil, default options are used.
func NewChecker(options *CheckerOptions) *Checker {
        if options == nil {
                options = DefaultCheckerOptions()
        }

        // Ensure we have at least one keyword
        if len(options.Keywords) == 0 {
                options.Keywords = DefaultKeywords
        }

        // Ensure lines to scan is positive
        if options.LinesToScan <= 0 {
                options.LinesToScan = 20
        }

        return &Checker{
                options: options,
        }
}

// Check examines a file for copyright notices.
// It reads the file and checks the first N lines for copyright keywords.
func (c *Checker) Check(ctx context.Context, filePath string) (*CheckResult, error) {
        // Check for context cancellation
        select {
        case <-ctx.Done():
                return nil, ctx.Err()
        default:
        }

        // Open the file
        file, err := os.Open(filePath)
        if err != nil {
                return &CheckResult{
                        FilePath:         filePath,
                        HasCopyright:     false,
                        DetectedKeywords: nil,
                        LineFound:        0,
                        Error:            fmt.Errorf("failed to open file: %w", err),
                }, nil
        }
        defer func() { _ = file.Close() }()

        // Read file content with line limit
        scanner := bufio.NewScanner(file)
        var lines []string
        lineCount := 0

        for scanner.Scan() && lineCount < c.options.LinesToScan {
                // Check for context cancellation periodically
                if lineCount%10 == 0 {
                        select {
                        case <-ctx.Done():
                                return nil, ctx.Err()
                        default:
                        }
                }

                line := scanner.Text()

                // Skip if line is too long (likely binary content)
                if len(line) > 10000 {
                        return &CheckResult{
                                FilePath:         filePath,
                                HasCopyright:     false,
                                DetectedKeywords: nil,
                                LineFound:        0,
                                Error:            fmt.Errorf("line too long (likely binary file)"),
                        }, nil
                }

                // Check if line contains valid UTF-8
                if !utf8.ValidString(line) {
                        return &CheckResult{
                                FilePath:         filePath,
                                HasCopyright:     false,
                                DetectedKeywords: nil,
                                LineFound:        0,
                                Error:            fmt.Errorf("invalid UTF-8 content (likely binary file)"),
                        }, nil
                }

                lines = append(lines, line)
                lineCount++
        }

        // Check for scanner errors
        if err := scanner.Err(); err != nil {
                return &CheckResult{
                        FilePath:         filePath,
                        HasCopyright:     false,
                        DetectedKeywords: nil,
                        LineFound:        0,
                        Error:            fmt.Errorf("failed to read file: %w", err),
                }, nil
        }

        // Convert lines to content bytes for checking
        content := []byte(strings.Join(lines, "\n"))
        result := c.CheckContent(content)
        result.FilePath = filePath

        return result, nil
}

// CheckContent examines content for copyright notices.
// This method is useful for testing or checking in-memory content.
func (c *Checker) CheckContent(content []byte) *CheckResult {
        result := &CheckResult{
                FilePath:         "",
                HasCopyright:     false,
                DetectedKeywords: make([]string, 0),
                LineFound:        0,
                Error:            nil,
        }

        // Handle empty content
        if len(content) == 0 {
                return result
        }

        // Check if content contains valid UTF-8
        if !utf8.Valid(content) {
                result.Error = fmt.Errorf("invalid UTF-8 content (likely binary)")
                return result
        }

        // Split content into lines
        scanner := bufio.NewScanner(bytes.NewReader(content))
        lineNum := 0
        firstMatchLine := 0
        keywordMatches := make(map[string]bool)

        for scanner.Scan() && lineNum < c.options.LinesToScan {
                lineNum++
                line := scanner.Text()

                // Convert to lowercase for case-insensitive matching
                lowerLine := strings.ToLower(line)

                // Check each keyword
                for _, keyword := range c.options.Keywords {
                        lowerKeyword := strings.ToLower(keyword)

                        if strings.Contains(lowerLine, lowerKeyword) {
                                // Mark this keyword as found
                                if !keywordMatches[keyword] {
                                        keywordMatches[keyword] = true
                                        result.DetectedKeywords = append(result.DetectedKeywords, keyword)
                                }

                                // Record the first line where any keyword was found
                                if firstMatchLine == 0 {
                                        firstMatchLine = lineNum
                                }
                        }
                }
        }

        // Determine if copyright was detected based on require_all setting
        if c.options.RequireAll {
                // All keywords must be present
                result.HasCopyright = len(keywordMatches) == len(c.options.Keywords)
        } else {
                // At least one keyword must be present
                result.HasCopyright = len(keywordMatches) > 0
        }

        result.LineFound = firstMatchLine

        return result
}

// TemplateData holds the data for copyright header template rendering
type TemplateData struct {

        Company string
        Author  string
        SPDX    string

// Generate generates a copyright header based on the configuration
func (g *HeaderGenerator) Generate() (string, error) {
        // Prepare template data
        data := TemplateData{
                Year:    g.getYear(),
                Company: g.config.Copyright.Company,
                Author:  g.config.Copyright.Author,
                SPDX:    g.config.Copyright.SPDXIdentifier,
        }

        // Get the template
        tmplText := g.config.Copyright.Template
        if tmplText == "" {
                return "", fmt.Errorf("copyright template is empty")
        }

        // Parse the template
        tmpl, err := template.New("copyright").Parse(tmplText)
        if err != nil {
                return "", fmt.Errorf("failed to parse copyright template: %w", err)
        }

        // Execute the template
        var buf bytes.Buffer
        if err := tmpl.Execute(&buf, data); err != nil {
                return "", fmt.Errorf("failed to execute copyright template: %w", err)
        }

        return buf.String(), nil
}

// ResolveYear handles special year values and year range calculation.
// It supports:
//   - "auto" - Returns the current year
//   - "range" - Returns a range from startYear to current year (e.g., "2020-2025")
//   - Specific year (e.g., "2025") - Returns as-is
//   - Year range (e.g., "2020-2025") - Returns as-is
//
// The startYear parameter is used when yearConfig is "range".
func ResolveYear(yearConfig string, startYear int) string {
        currentYear := time.Now().Year()

        // Normalize the year config
        yearConfig = strings.TrimSpace(yearConfig)

        // Handle "auto" - return current year
        if yearConfig == "auto" || yearConfig == "" {
                return strconv.Itoa(currentYear)
        }

        // Handle "range" - return range from startYear to current year
        if yearConfig == "range" {
                if startYear <= 0 || startYear > currentYear {
                        // If invalid start year, just use current year
                        return strconv.Itoa(currentYear)
                }

                if startYear == currentYear {
                        // Same year, no range needed
                        return strconv.Itoa(currentYear)
                }

                return fmt.Sprintf("%d-%d", startYear, currentYear)
        }

        // Otherwise, return the year config as-is
        // This handles explicit years (e.g., "2025") and explicit ranges (e.g., "2020-2025")
        return yearConfig
}

// ParseYear parses a year string and returns the year as an integer.
// It handles:
//   - Single year: "2025" -> 2025
//   - Year range: "2020-2025" -> 2020 (returns the start year)
//   - "auto": returns current year
//
// Returns an error if the year string is invalid.
func ParseYear(yearStr string) (int, error) {
        yearStr = strings.TrimSpace(yearStr)

        if yearStr == "" || yearStr == "auto" {
                return time.Now().Year(), nil
        }

        // Check if it's a range (YYYY-YYYY)
        if strings.Contains(yearStr, "-") {
                parts := strings.Split(yearStr, "-")
                if len(parts) != 2 {
                        return 0, fmt.Errorf("invalid year range format: %s", yearStr)
                }

                // Parse the start year
                startYear, err := strconv.Atoi(strings.TrimSpace(parts[0]))
                if err != nil {
                        return 0, fmt.Errorf("invalid start year in range: %s", parts[0])
                }

                return startYear, nil
        }

        // Parse as a single year
        year, err := strconv.Atoi(yearStr)
        if err != nil {
                return 0, fmt.Errorf("invalid year format: %s", yearStr)
        }

        if year < 1900 || year > 2200 {
                return 0, fmt.Errorf("year out of valid range (1900-2200): %d", year)
        }

        return year, nil
}

// IsBinary determines if a file is binary by checking:
// 1. File extension
// 2. Magic number signatures
// 3. Null byte detection in first 8KB
func IsBinary(path string) (bool, error) {
        // Quick check: common binary extensions
        ext := filepath.Ext(path)
        if commonBinaryExtensions[ext] {
                return true, nil
        }

        // Open the file
        f, err := os.Open(path)
        if err != nil {
                return false, err
        }
        defer func() { _ = f.Close() }()

        // Read first bufferSize bytes
        buffer := make([]byte, bufferSize)
        n, err := f.Read(buffer)
        if err != nil && err != io.EOF {
                return false, err
        }
        buffer = buffer[:n]

        // Empty files are not binary
        if n == 0 {
                return false, nil
        }

        // Check for magic number signatures
        if hasBinarySignature(buffer) {
                return true, nil
        }

        // Check for null bytes (strong indicator of binary content)
        if bytes.Contains(buffer, []byte{0}) {
                return true, nil
        }

        // Check for high ratio of non-printable characters
        nonPrintable := 0
        for _, b := range buffer {
                // Count non-ASCII printable characters
                // ASCII printable: 32-126, plus common whitespace: 9 (tab), 10 (LF), 13 (CR)
                if (b < 32 || b > 126) && b != 9 && b != 10 && b != 13 {
                        nonPrintable++
                }
        }

        // If more than 30% of characters are non-printable, likely binary
        if float64(nonPrintable)/float64(n) > 0.3 {
                return true, nil
        }

        return false, nil
}

// hasBinarySignature checks if the buffer starts with a known binary signature
func hasBinarySignature(buffer []byte) bool {
        for _, signature := range commonBinarySignatures {
                if len(buffer) >= len(signature) {
                        if bytes.HasPrefix(buffer, signature) {
                                return true
                        }
                }
        }
        return false
}

// GitignoreMatcher handles gitignore pattern matching
type GitignoreMatcher struct {
        rootPath string
        matchers map[string]*ignore.GitIgnore // path -> matcher for nested .gitignore files
        global   *ignore.GitIgnore            // global patterns
}

// NewGitignoreMatcher creates a new gitignore matcher for the given root path
func NewGitignoreMatcher(rootPath string) (*GitignoreMatcher, error) {
        absRoot, err := filepath.Abs(rootPath)
        if err != nil {
                return nil, fmt.Errorf("failed to get absolute path: %w", err)
        }

        gm := &GitignoreMatcher{
                rootPath: absRoot,
                matchers: make(map[string]*ignore.GitIgnore),
        }

        // Load root .gitignore if it exists
        if err := gm.loadGitignoreFile(absRoot); err != nil {
                // If .gitignore doesn't exist, that's okay
                if !os.IsNotExist(err) {
                        return nil, err
                }
        }

        // Initialize global matcher with empty patterns (will be populated via AddPatterns)
        gm.global = ignore.CompileIgnoreLines()

        return gm, nil
}

// LoadNestedGitignore loads a .gitignore file from a subdirectory
func (g *GitignoreMatcher) LoadNestedGitignore(dirPath string) error {
        return g.loadGitignoreFile(dirPath)
}

// loadGitignoreFile loads a .gitignore file from a directory
func (g *GitignoreMatcher) loadGitignoreFile(dirPath string) error {
        gitignorePath := filepath.Join(dirPath, ".gitignore")

        // Check if .gitignore exists
        info, err := os.Stat(gitignorePath)
        if err != nil {
                return err
        }

        // Don't process if it's not a regular file
        if !info.Mode().IsRegular() {
                return nil
        }

        // Compile the gitignore file
        matcher, err := ignore.CompileIgnoreFile(gitignorePath)
        if err != nil {
                return fmt.Errorf("failed to compile gitignore file %s: %w", gitignorePath, err)
        }

        // Store the matcher for this directory
        g.matchers[dirPath] = matcher

        return nil
}

// Match checks if a path should be ignored based on gitignore rules
func (g *GitignoreMatcher) Match(path string) bool {
        // Get absolute path
        absPath, err := filepath.Abs(path)
        if err != nil {
                return false
        }

        // Get relative path from root
        relPath, err := filepath.Rel(g.rootPath, absPath)
        if err != nil {
                return false
        }

        // Normalize path separators for gitignore matching
        relPath = filepath.ToSlash(relPath)

        // Check global patterns first
        if g.global != nil && g.global.MatchesPath(relPath) {
                return true
        }

        // Check each .gitignore from root to the file's directory
        segments := strings.Split(filepath.Dir(absPath), string(filepath.Separator))

        for i := len(strings.Split(g.rootPath, string(filepath.Separator))); i <= len(segments); i++ {
                checkDir := strings.Join(segments[:i], string(filepath.Separator))
                if checkDir == "" {
                        continue
                }

                // Get relative path from this directory
                relFromDir, err := filepath.Rel(checkDir, absPath)
                if err != nil {
                        continue
                }
                relFromDir = filepath.ToSlash(relFromDir)

                // Check if this directory has a gitignore matcher
                if matcher, exists := g.matchers[checkDir]; exists {
                        if matcher.MatchesPath(relFromDir) {
                                return true
                        }
                        // Also check with just the filename for patterns like "*.log"
                        if matcher.MatchesPath(filepath.Base(absPath)) {
                                return true
                        }
                }
        }

        return false
}

// AddPatterns adds custom ignore patterns to the global matcher
func (g *GitignoreMatcher) AddPatterns(patterns []string) error {
        if len(patterns) == 0 {
                return nil
        }

        // Compile patterns
        g.global = ignore.CompileIgnoreLines(patterns...)

        return nil
}

// ShouldIgnoreDir checks if a directory should be ignored
// This is useful for early pruning of directory trees
func (g *GitignoreMatcher) ShouldIgnoreDir(dirPath string) bool {
        // Always ignore .git directories
        if filepath.Base(dirPath) == ".git" {
                return true
        }

        // Check gitignore patterns
        return g.Match(dirPath)
}

// FilterByExtension returns files matching the given extensions
func FilterByExtension(files []FileInfo, extensions ...string) []FileInfo {
        if len(extensions) == 0 {
                return files
        }

        // Normalize extensions
        normalizedExts := make(map[string]bool)
        for _, ext := range extensions {
                if ext != "" && !strings.HasPrefix(ext, ".") {
                        ext = "." + ext
                }
                normalizedExts[strings.ToLower(ext)] = true
        }

        var result []FileInfo
        for _, file := range files {
                if normalizedExts[strings.ToLower(file.Extension)] {
                        result = append(result, file)
                }
        }

        return result
}

// FindLargestFiles returns the N largest files
func FindLargestFiles(files []FileInfo, n int) []FileInfo {
        if n <= 0 || len(files) == 0 {
                return nil
        }

        // Make a copy to avoid modifying the original slice
        sorted := make([]FileInfo, len(files))
        copy(sorted, files)

        // Simple selection sort for finding top N
        for i := 0; i < n && i < len(sorted); i++ {
                maxIdx := i
                for j := i + 1; j < len(sorted); j++ {
                        if sorted[j].Size > sorted[maxIdx].Size {
                                maxIdx = j
                        }
                }
                sorted[i], sorted[maxIdx] = sorted[maxIdx], sorted[i]
        }

        if n > len(sorted) {
                n = len(sorted)
        }
        return sorted[:n]
}

}

// MatchPattern matches a glob pattern against a path
// Supports:
// - * matches any characters except /
// - ** matches any characters including /
// - ? matches a single character
// - [abc] character classes
func MatchPattern(pattern, path string) (bool, error) {
        // Normalize path separators
        pattern = filepath.ToSlash(pattern)
        path = filepath.ToSlash(path)

        // Handle ** patterns specially
        if strings.Contains(pattern, "**") {
                return matchDoubleStarPattern(pattern, path)
        }

        // Use filepath.Match for simple patterns
        matched, err := filepath.Match(pattern, path)
        if err != nil {
                return false, err
        }

        // Also try matching just the base name
        if !matched {
                matched, err = filepath.Match(pattern, filepath.Base(path))
        }

        return matched, err
}

// MatchAnyPattern returns true if the path matches any of the provided patterns
func MatchAnyPattern(patterns []string, path string) bool {
        for _, pattern := range patterns {
                matched, err := MatchPattern(pattern, path)
                if err != nil {
                        // If pattern is invalid, skip it
                        continue
                }
                if matched {
                        return true
                }
        }
        return false
}

// matchDoubleStarPattern handles patterns containing **
func matchDoubleStarPattern(pattern, path string) (bool, error) {
        // Split pattern by **
        parts := strings.Split(pattern, "**")

        // If pattern starts with **, match any prefix
        if strings.HasPrefix(pattern, "**") {
                // Remove leading **
                if len(parts) > 1 && parts[0] == "" {
                        parts = parts[1:]
                }
        }

        // If pattern ends with **, match any suffix
        if strings.HasSuffix(pattern, "**") {
                // Remove trailing **
                if len(parts) > 0 && parts[len(parts)-1] == "" {
                        parts = parts[:len(parts)-1]
                }
        }

        // Handle single ** (matches everything)
        if len(parts) == 0 || (len(parts) == 1 && parts[0] == "") {
                return true, nil
        }

        // For multiple parts, we need to match them in sequence
        return matchPatternParts(parts, path)
}

// matchPatternParts matches a path against pattern parts separated by **
func matchPatternParts(parts []string, path string) (bool, error) {
        if len(parts) == 0 {
                return true, nil
        }

        // Clean up empty parts and trim slashes
        var cleanParts []string
        for _, part := range parts {
                part = strings.Trim(part, "/")
                if part != "" {
                        cleanParts = parts
                }
        }

        if len(cleanParts) == 0 {
                return true, nil
        }

        // For simplicity with ** patterns, check if the path contains all parts in order
        currentPath := path
        for i, part := range cleanParts {
                part = strings.Trim(part, "/")
                if part == "" {
                        continue
                }

                // For the first part, check if it matches the beginning
                if i == 0 && strings.HasPrefix(path, part) {
                        // Check if this is an exact segment match
                        matched, err := matchPathSegment(part, currentPath, true)
                        if err != nil {
                                return false, err
                        }
                        if matched {
                                // Move past this part
                                idx := strings.Index(currentPath, "/")
                                if idx > 0 {
                                        currentPath = currentPath[idx+1:]
                                } else {
                                        currentPath = ""
                                }
                                continue
                        }
                }

                // For middle parts, find them anywhere
                matched := false
                segments := strings.Split(currentPath, "/")
                for j, segment := range segments {
                        segmentMatched, err := filepath.Match(part, segment)
                        if err != nil {
                                return false, err
                        }
                        if segmentMatched {
                                matched = true
                                // Move past this segment
                                if j+1 < len(segments) {
                                        currentPath = strings.Join(segments[j+1:], "/")
                                } else {
                                        currentPath = ""
                                }
                                break
                        }
                }

                if !matched {
                        return false, nil
                }
        }

        return true, nil
}

// matchPathSegment checks if a pattern matches a segment at the start of the path
func matchPathSegment(pattern, path string, atStart bool) (bool, error) {
        segments := strings.Split(path, "/")
        if len(segments) == 0 {
                return false, nil
        }

        if atStart {
                return filepath.Match(pattern, segments[0])
        }

        // Check any segment
        for _, segment := range segments {
                matched, err := filepath.Match(pattern, segment)
                if err != nil {
                        return false, err
                }
                if matched {
                        return true, nil
                }
        }

        return false, nil
}

// Scanner walks a directory tree and discovers files according to configured options
type Scanner struct {
        rootPath  string
        gitignore *GitignoreMatcher
        options   *ScanOptions
}

// NewScanner creates a new file scanner for the given root path
func NewScanner(rootPath string, options *ScanOptions) (*Scanner, error) {
        // Get absolute path
        absRoot, err := filepath.Abs(rootPath)
        if err != nil {
                return nil, fmt.Errorf("failed to get absolute path: %w", err)
        }

        // Verify root path exists and is a directory
        info, err := os.Stat(absRoot)
        if err != nil {
                return nil, fmt.Errorf("failed to stat root path: %w", err)
        }
        if !info.IsDir() {
                return nil, fmt.Errorf("root path is not a directory: %s", absRoot)
        }

        // Use default options if none provided
        if options == nil {
                options = DefaultScanOptions()
        }

        // Initialize gitignore matcher
        gitignoreMatcher, err := NewGitignoreMatcher(absRoot)
        if err != nil {
                return nil, fmt.Errorf("failed to initialize gitignore matcher: %w", err)
        }

        // Add custom exclude patterns to gitignore
        if len(options.ExcludePatterns) > 0 {
                if err := gitignoreMatcher.AddPatterns(options.ExcludePatterns); err != nil {
                        return nil, fmt.Errorf("failed to add exclude patterns: %w", err)
                }
        }

        return &Scanner{
                rootPath:  absRoot,
                gitignore: gitignoreMatcher,
                options:   options,
        }, nil
}

// Scan walks the directory tree and returns discovered files
func (s *Scanner) Scan(ctx context.Context) (*ScanResult, error) {
        result := &ScanResult{
                Files:        make([]FileInfo, 0),
                SkippedFiles: make([]string, 0),
                Errors:       make([]error, 0),
        }

        // Walk the directory tree
        err := filepath.WalkDir(s.rootPath, func(path string, d fs.DirEntry, err error) error {
                // Check for context cancellation
                select {
                case <-ctx.Done():
                        return ctx.Err()
                default:
                }

                // Handle errors from WalkDir
                if err != nil {
                        // Log permission errors but continue
                        if os.IsPermission(err) {
                                result.Errors = append(result.Errors, fmt.Errorf("permission denied: %s", path))
                                return nil
                        }
                        return err
                }

                // Get relative path
                relPath, err := filepath.Rel(s.rootPath, path)
                if err != nil {
                        result.Errors = append(result.Errors, fmt.Errorf("failed to get relative path for %s: %w", path, err))
                        return nil
                }

                // Handle directories
                if d.IsDir() {
                        // Skip .git directory always
                        if d.Name() == ".git" {
                                return filepath.SkipDir
                        }

                        // Skip hidden directories if configured
                        if s.options.SkipHidden && strings.HasPrefix(d.Name(), ".") && path != s.rootPath {
                                result.SkippedFiles = append(result.SkippedFiles, relPath)
                                return filepath.SkipDir
                        }

                        // Check gitignore patterns
                        if s.gitignore.ShouldIgnoreDir(path) {
                                result.SkippedFiles = append(result.SkippedFiles, relPath)
                                return filepath.SkipDir
                        }

                        // Try to load nested .gitignore
                        _ = s.gitignore.LoadNestedGitignore(path)

                        return nil
                }

                // Handle symlinks
                if d.Type()&os.ModeSymlink != 0 {
                        if !s.options.FollowSymlinks {
                                result.SkippedFiles = append(result.SkippedFiles, relPath)
                                return nil
                        }
                        // If following symlinks, resolve and check if it's a file
                        realPath, err := filepath.EvalSymlinks(path)
                        if err != nil {
                                result.Errors = append(result.Errors, fmt.Errorf("failed to resolve symlink %s: %w", path, err))
                                return nil
                        }
                        info, err := os.Stat(realPath)
                        if err != nil {
                                result.Errors = append(result.Errors, fmt.Errorf("failed to stat symlink target %s: %w", realPath, err))
                                return nil
                        }
                        if info.IsDir() {
                                // Don't follow directory symlinks even if FollowSymlinks is true
                                result.SkippedFiles = append(result.SkippedFiles, relPath)
                                return nil
                        }
                }

                // Process files only
                if !d.Type().IsRegular() {
                        return nil
                }

                // Skip hidden files if configured
                if s.options.SkipHidden && strings.HasPrefix(d.Name(), ".") {
                        result.SkippedFiles = append(result.SkippedFiles, relPath)
                        return nil
                }

                // Check gitignore patterns
                if s.gitignore.Match(path) {
                        result.SkippedFiles = append(result.SkippedFiles, relPath)
                        return nil
                }

                // Check excluded extensions
                ext := filepath.Ext(path)
                if s.isExcludedExtension(ext) {
                        result.SkippedFiles = append(result.SkippedFiles, relPath)
                        return nil
                }

                // Check include patterns (if specified, only include matching files)
                if len(s.options.IncludePatterns) > 0 {
                        if !MatchAnyPattern(s.options.IncludePatterns, relPath) {
                                result.SkippedFiles = append(result.SkippedFiles, relPath)
                                return nil
                        }
                }

                // Get file info
                info, err := d.Info()
                if err != nil {
                        result.Errors = append(result.Errors, fmt.Errorf("failed to get file info for %s: %w", path, err))
                        return nil
                }

                // Check if binary
                isBinary, err := IsBinary(path)
                if err != nil {
                        // If we can't determine, assume it's binary and skip
                        result.Errors = append(result.Errors, fmt.Errorf("failed to check if binary %s: %w", path, err))
                        result.SkippedFiles = append(result.SkippedFiles, relPath)
                        return nil
                }

                // Create FileInfo
                fileInfo := FileInfo{
                        Path:         path,
                        RelativePath: relPath,
                        Extension:    ext,
                        Filename:     d.Name(),
                        Size:         info.Size(),
                        IsBinary:     isBinary,
                }

                result.Files = append(result.Files, fileInfo)
                return nil
        })

        if err != nil && err != context.Canceled && err != context.DeadlineExceeded {
                return nil, fmt.Errorf("failed to walk directory tree: %w", err)
        }

        // If context was canceled, include that as an error
        if err == context.Canceled || err == context.DeadlineExceeded {
                result.Errors = append(result.Errors, err)
        }

        return result, nil
}

// isExcludedExtension checks if a file extension is in the excluded list
func (s *Scanner) isExcludedExtension(ext string) bool {
        if len(s.options.ExcludedExtensions) == 0 {
                return false
        }

        // Normalize extension (ensure it starts with .)
        if ext != "" && !strings.HasPrefix(ext, ".") {
                ext = "." + ext
        }

        for _, excluded := range s.options.ExcludedExtensions {
                // Normalize excluded extension
                if excluded != "" && !strings.HasPrefix(excluded, ".") {
                        excluded = "." + excluded
                }
                if strings.EqualFold(ext, excluded) {
                        return true
                }
        }

        return false
}

// GetRootPath returns the scanner's root path
func (s *Scanner) GetRootPath() string {
        return s.rootPath
}

// Package language provides language-specific copyright header formatting.
//
// This package supports automatic detection of programming languages and file types
// based on file extensions and specific filenames, and provides language-appropriate
// comment formatting for copyright headers.
//
// Supported Languages:
//   - Go (.go)
//   - JavaScript/TypeScript (.js, .ts, .jsx, .tsx, .mjs, .cjs)
//   - Java (.java)
//   - C# (.cs)
//   - Rust (.rs)
//   - C/C++ (.c, .h, .cpp, .hpp, .cc, .cxx)
//   - Python (.py, .pyw)
//   - Ruby (.rb, .rake, Rakefile, Gemfile)
//   - Shell (.sh, .bash, .zsh, .fish)
//   - YAML/TOML (.yaml, .yml, .toml)
//   - HTML/XML (.html, .htm, .xml, .svg)
//   - CSS (.css, .scss, .sass, .less)
//   - SQL (.sql)
//   - Markdown (.md, .markdown) - no comments supported
//   - JSON (.json) - no comments supported
//   - Protocol Buffers (.proto)
//   - Makefile (Makefile, makefile, GNUmakefile)
//   - Dockerfile (Dockerfile, Containerfile)
//   - Terraform (.tf, .tfvars, .hcl)
//
// Usage:
//
//        registry := language.GetRegistry()
//        lang := registry.GetLanguage("main.go")
//        if lang != nil && lang.CanHaveComments() {
//            copyright := "Copyright 2025 Lean Security Co\nSPDX-License-Identifier: MIT"
//            header := lang.FormatHeader(copyright)
//            // header contains properly formatted comment
//        }
//
// Special Features:
//   - Preserves shebang lines (#!) for scripts
//   - Preserves Python encoding declarations
//   - Preserves Ruby encoding declarations
//   - Preserves XML declarations
//   - Case-insensitive filename matching for special files
package language

import (
        "bytes"
        "strings"
)

// CommentStyle defines the type of comment syntax a language uses
type CommentStyle int

const (
        // CommentStyleNone indicates the file type doesn't support comments
        CommentStyleNone CommentStyle = iota
        // CommentStyleLine indicates line comments (e.g., //, #)
        CommentStyleLine
        // CommentStyleBlock indicates block comments (e.g., /* */)
        CommentStyleBlock
        // CommentStyleHTML indicates HTML-style comments (e.g., <!-- -->)
        CommentStyleHTML
        // CommentStyleSQL indicates SQL-style comments (e.g., --)
        CommentStyleSQL
)

// Language defines the interface for language-specific copyright header handling
type Language interface {
        // Name returns the human-readable name of the language
        Name() string

        // Extensions returns the file extensions this language handles (e.g., ".go")
        Extensions() []string

        // Filenames returns specific filenames this language handles (e.g., "Makefile")
        Filenames() []string

        // CanHaveComments returns whether this file type supports comments
        CanHaveComments() bool

        // FormatHeader formats the copyright text as a comment block for this language
        FormatHeader(copyrightText string) string

        // PreserveLines returns lines that should be preserved before the copyright
        // (e.g., shebang lines, encoding declarations)
        PreserveLines(content []byte) []string

        // InsertPosition returns the byte offset where the copyright should be inserted
        InsertPosition(content []byte) int
}

// BaseLanguage provides common implementation for language handlers
type BaseLanguage struct {
        name         string
        extensions   []string
        filenames    []string
        commentStyle CommentStyle
        linePrefix   string // For line and hash comments
        blockStart   string // For block comments
        blockPrefix  string // For block comments (continuation)
        blockEnd     string // For block comments
}

// Name returns the language name
func (b *BaseLanguage) Name() string {
        return b.name
}

// Extensions returns the file extensions
func (b *BaseLanguage) Extensions() []string {
        return b.extensions
}

// Filenames returns specific filenames
func (b *BaseLanguage) Filenames() []string {
        return b.filenames
}

// CanHaveComments returns whether the language supports comments
func (b *BaseLanguage) CanHaveComments() bool {
        return b.commentStyle != CommentStyleNone
}

// FormatHeader formats the copyright text as a comment
func (b *BaseLanguage) FormatHeader(copyrightText string) string {
        lines := strings.Split(strings.TrimSpace(copyrightText), "\n")

        switch b.commentStyle {
        case CommentStyleLine:
                // Line comments like // or #
                var formatted []string
                for _, line := range lines {
                        if line == "" {
                                formatted = append(formatted, b.linePrefix)
                        } else {
                                formatted = append(formatted, b.linePrefix+" "+line)
                        }
                }
                return strings.Join(formatted, "\n") + "\n"

        case CommentStyleBlock, CommentStyleHTML:
                // Block comments like /* */ or <!-- -->
                var formatted []string
                formatted = append(formatted, b.blockStart)
                for _, line := range lines {
                        if line == "" {
                                formatted = append(formatted, b.blockPrefix)
                        } else {
                                formatted = append(formatted, b.blockPrefix+" "+line)
                        }
                }
                formatted = append(formatted, b.blockEnd)
                return strings.Join(formatted, "\n") + "\n"

        case CommentStyleSQL:
                // SQL-style comments with --
                var formatted []string
                for _, line := range lines {
                        if line == "" {
                                formatted = append(formatted, b.linePrefix)
                        } else {
                                formatted = append(formatted, b.linePrefix+" "+line)
                        }
                }
                return strings.Join(formatted, "\n") + "\n"

        case CommentStyleNone:
                // No comment support
                return ""

        default:
                return ""
        }
}

// PreserveLines returns lines to preserve (default: none)
func (b *BaseLanguage) PreserveLines(content []byte) []string {
        return nil
}

// InsertPosition returns the position to insert the copyright
// Note: This method cannot be implemented in BaseLanguage because it needs to call
// the overridden PreserveLines method from the specific language implementation.
// Each language that overrides PreserveLines must also override InsertPosition.
func (b *BaseLanguage) InsertPosition(content []byte) int {
        // For languages without special preserved lines, insert at the beginning
        return 0
}

// hasShebang checks if the content starts with a shebang line
func hasShebang(content []byte) bool {
        return bytes.HasPrefix(content, []byte("#!"))
}

// getShebangLine extracts the shebang line if present
func getShebangLine(content []byte) string {
        if !hasShebang(content) {
                return ""
        }

        idx := bytes.IndexByte(content, '\n')
        if idx == -1 {
                return string(content)
        }
        return string(content[:idx+1])
}

// hasPythonEncoding checks if a line is a Python encoding declaration
func hasPythonEncoding(line string) bool {
        line = strings.TrimSpace(line)
        if !strings.HasPrefix(line, "#") {
                return false
        }

        // Python encoding: # -*- coding: utf-8 -*-
        // or: # coding: utf-8
        // or: # coding=utf-8
        return strings.Contains(line, "coding:") ||
                strings.Contains(line, "coding=") ||
                strings.Contains(line, "-*- coding")
}

// calculateInsertPosition calculates byte position after preserved lines
func calculateInsertPosition(preserved []string) int {
        position := 0
        for _, line := range preserved {
                position += len(line)
        }
        return position
}

// PreserveLines handles XML declaration for XML files
func (h *HTML) PreserveLines(content []byte) []string {
        var preserve []string

        // Check for XML declaration (<?xml version="1.0" ?>)
        if bytes.HasPrefix(bytes.TrimSpace(content), []byte("<?xml")) {
                idx := bytes.Index(content, []byte("?>"))
                if idx != -1 {
                        // Find the newline after the declaration
                        endIdx := idx + 2
                        newlineIdx := bytes.IndexByte(content[endIdx:], '\n')
                        if newlineIdx != -1 {
                                preserve = append(preserve, string(content[:endIdx+newlineIdx+1]))
                        } else {
                                preserve = append(preserve, string(content[:endIdx])+"\n")
                        }
                }
        }

        // Check for DOCTYPE declaration
        trimmed := bytes.TrimSpace(content)
        if bytes.HasPrefix(trimmed, []byte("<!DOCTYPE")) || bytes.HasPrefix(trimmed, []byte("<!doctype")) {
                // Find the position in original content
                doctypeStart := bytes.Index(content, []byte("<!DOCTYPE"))
                if doctypeStart == -1 {
                        doctypeStart = bytes.Index(content, []byte("<!doctype"))
                }

                if doctypeStart != -1 {
                        remaining := content[doctypeStart:]
                        idx := bytes.IndexByte(remaining, '>')
                        if idx != -1 {
                                // Find the newline after the declaration
                                endIdx := doctypeStart + idx + 1
                                newlineIdx := bytes.IndexByte(content[endIdx:], '\n')
                                if newlineIdx != -1 {
                                        doctype := string(content[:endIdx+newlineIdx+1])
                                        // Avoid duplicates
                                        if len(preserve) == 0 || !strings.Contains(preserve[0], "DOCTYPE") {
                                                preserve = append(preserve, doctype)
                                        }
                                }
                        }
                }
        }

        return preserve
}

// InsertPosition returns the byte offset where copyright should be inserted
func (h *HTML) InsertPosition(content []byte) int {
        return calculateInsertPosition(h.PreserveLines(content))
}

// Registry manages the mapping of file extensions and names to language handlers
type Registry struct {
        mu                 sync.RWMutex
        extensionMap       map[string]Language
        filenameMap        map[string]Language
        caseInsensitiveMap map[string]Language // For case-insensitive filename matching
}

var defaultRegistry *Registry
var once sync.Once

// GetRegistry returns the singleton registry instance
func GetRegistry() *Registry {
        once.Do(func() {
                defaultRegistry = &Registry{
                        extensionMap:       make(map[string]Language),
                        filenameMap:        make(map[string]Language),
                        caseInsensitiveMap: make(map[string]Language),
                }
                registerAllLanguages()
        })
        return defaultRegistry
}

// Register adds a language to the registry
func (r *Registry) Register(lang Language) {
        r.mu.Lock()
        defer r.mu.Unlock()

        // Register extensions
        for _, ext := range lang.Extensions() {
                r.extensionMap[ext] = lang
        }

        // Register specific filenames
        for _, filename := range lang.Filenames() {
                r.filenameMap[filename] = lang
                // Also register lowercase version for case-insensitive matching
                r.caseInsensitiveMap[strings.ToLower(filename)] = lang
        }
}

// GetLanguage returns the language handler for a given filename
func (r *Registry) GetLanguage(filename string) Language {
        r.mu.RLock()
        defer r.mu.RUnlock()

        // Extract the base filename
        base := filepath.Base(filename)

        // First, check exact filename match
        if lang, ok := r.filenameMap[base]; ok {
                return lang
        }

        // Check case-insensitive filename match
        if lang, ok := r.caseInsensitiveMap[strings.ToLower(base)]; ok {
                return lang
        }

        // Then check by extension
        ext := filepath.Ext(filename)
        if lang, ok := r.extensionMap[ext]; ok {
                return lang
        }

        // Check for double extensions (e.g., .test.js)
        if len(ext) > 0 {
                // Remove the extension and try again
                nameWithoutExt := strings.TrimSuffix(base, ext)
                secondExt := filepath.Ext(nameWithoutExt)
                if len(secondExt) > 0 {
                        combinedExt := secondExt + ext
                        if lang, ok := r.extensionMap[combinedExt]; ok {
                                return lang
                        }
                }
        }

        return nil
}

// SupportedExtensions returns all registered file extensions
func (r *Registry) SupportedExtensions() []string {
        r.mu.RLock()
        defer r.mu.RUnlock()

        exts := make([]string, 0, len(r.extensionMap))
        for ext := range r.extensionMap {
                exts = append(exts, ext)
        }
        return exts
}

// SupportedFilenames returns all registered specific filenames
func (r *Registry) SupportedFilenames() []string {
        r.mu.RLock()
        defer r.mu.RUnlock()

        names := make([]string, 0, len(r.filenameMap))
        for name := range r.filenameMap {
                names = append(names, name)
        }
        return names
}

// registerAllLanguages registers all built-in language handlers

        registry := defaultRegistry

        // Register all language handlers
        registry.Register(NewGoLang())
        registry.Register(NewJavaScript())
        registry.Register(NewJava())

        registry.Register(NewYAML())
        registry.Register(NewHTML())
        registry.Register(NewCSS())
        registry.Register(NewSQL())
        registry.Register(NewMarkdown())
        registry.Register(NewJSON())
        registry.Register(NewProtocolBuffers())
        registry.Register(NewMakefile())
        registry.Register(NewDockerfile())
        registry.Register(NewTerraform())

var commonLicenseNames = []string{
        "LICENSE",
        "LICENSE.txt",
        "LICENSE.md",
        "COPYING",
        "COPYING.txt",
        "License",
        "license",
        "license.txt",
}

// Exists checks if a LICENSE file exists in the directory
func (m *Manager) Exists(dir string) bool {
        _, exists := m.FindLicense(dir)
        return exists
}

// FindLicense searches for a LICENSE file in the directory
// Returns the path to the LICENSE file and whether it was found
func (m *Manager) FindLicense(dir string) (string, bool) {
        for _, name := range commonLicenseNames {
                path := filepath.Join(dir, name)
                if _, err := os.Stat(path); err == nil {
                        return path, true
                }
        }
        return "", false
}

// Generate generates the license content based on the license type
func (m *Manager) Generate() (string, error) {
        // Handle custom license type
        if m.licenseType == "custom" {
                if m.customText == "" {
                        return "", fmt.Errorf("custom license text is required for custom license type")
                }
                return m.customText, nil
        }

        // Get the template
        templateContent, err := GetTemplate(m.licenseType)
        if err != nil {
                return "", err
        }

        // Parse and execute the template
        tmpl, err := template.New("license").Parse(templateContent)
        if err != nil {
                return "", fmt.Errorf("failed to parse license template: %w", err)
        }

        var buf bytes.Buffer
        if err := tmpl.Execute(&buf, m.data); err != nil {
                return "", fmt.Errorf("failed to execute license template: %w", err)
        }

        return buf.String(), nil
}

// Create creates a LICENSE file in the specified directory
func (m *Manager) Create(dir string) error {
        // Check if a LICENSE file already exists
        if existingPath, exists := m.FindLicense(dir); exists {
                return fmt.Errorf("license file already exists: %s", existingPath)
        }

        // Generate the license content
        content, err := m.Generate()
        if err != nil {
                return err
        }

        // Create the LICENSE file
        licensePath := filepath.Join(dir, "LICENSE")
        if err := os.WriteFile(licensePath, []byte(content), 0644); err != nil {
                return fmt.Errorf("failed to write LICENSE file: %w", err)
        }

        return nil
}

// GetTemplate returns template content for a license type

        filename := fmt.Sprintf("templates/%s.txt", licenseType)
        content, err := licenseTemplates.ReadFile(filename)
        if err != nil {
                return "", fmt.Errorf("license template not found: %s", licenseType)
        }

}

// ListTemplates returns available license types
func ListTemplates() []string {
        entries, err := licenseTemplates.ReadDir("templates")
        if err != nil {
                return []string{}
        }

        var templates []string
        for _, entry := range entries {
                if entry.IsDir() {
                        continue
                }
                name := entry.Name()
                if strings.HasSuffix(name, ".txt") {
                        // Remove .txt extension
                        templates = append(templates, strings.TrimSuffix(name, filepath.Ext(name)))
                }
        }

        sort.Strings(templates)
        return templates
}

// Processor orchestrates the copyright management workflow
type Processor struct {
        config    *config.Config
        scanner   *files.Scanner
        checker   *copyright.Checker
        generator *copyright.HeaderGenerator
        writer    *Writer
        registry  *language.Registry
        license   *license.Manager
}

// ProcessorOptions configures processor behavior
type ProcessorOptions struct {
        DryRun bool // If true, don't modify files
        Backup bool // If true, create .bak files before modifying
}

// NewProcessor creates a new processor with the given configuration
func NewProcessor(cfg *config.Config) (*Processor, error) {
        if cfg == nil {
                return nil, fmt.Errorf("config cannot be nil")
        }

        // Create checker options from config
        checkerOpts := &copyright.CheckerOptions{
                LinesToScan: cfg.Detection.LinesToScan,
                Keywords:    cfg.Detection.Keywords,
                RequireAll:  cfg.Detection.RequireAll,
        }

        // Create license manager with config values
        licenseData := &license.LicenseData{
                Year:    cfg.Copyright.Year,
                Company: cfg.Copyright.Company,
                Author:  cfg.Copyright.Author,
        }

        return &Processor{
                config:    cfg,
                checker:   copyright.NewChecker(checkerOpts),
                generator: copyright.NewHeaderGenerator(cfg),
                registry:  language.GetRegistry(),
                license:   license.NewManager(cfg.License.Type, cfg.License.CustomText, licenseData),
        }, nil
}

// Process runs the copyright check/fix workflow on the given root path
func (p *Processor) Process(ctx context.Context, rootPath string, mode Mode) (*ProcessResult, error) {
        return p.ProcessWithOptions(ctx, rootPath, mode, &ProcessorOptions{})
}

// ProcessWithOptions runs the copyright check/fix workflow with custom options
func (p *Processor) ProcessWithOptions(ctx context.Context, rootPath string, mode Mode, opts *ProcessorOptions) (*ProcessResult, error) {
        // Initialize result
        result := &ProcessResult{
                FileResults: make([]FileResult, 0),
        }

        // Get absolute path
        absPath, err := filepath.Abs(rootPath)
        if err != nil {
                return nil, fmt.Errorf("failed to get absolute path: %w", err)
        }

        // Verify path exists
        if _, err := os.Stat(absPath); err != nil {
                return nil, fmt.Errorf("path does not exist: %w", err)
        }

        // Create scanner options from config
        scanOpts := &files.ScanOptions{
                ExcludePatterns:    p.config.Scan.ExcludePatterns,
                IncludePatterns:    p.config.Scan.IncludePatterns,
                ExcludedExtensions: p.config.Scan.ExcludedExtensions,
                SkipHidden:         p.config.Scan.SkipHidden,
                FollowSymlinks:     p.config.Scan.FollowSymlinks,
        }

        // Create scanner
        scanner, err := files.NewScanner(absPath, scanOpts)
        if err != nil {
                return nil, fmt.Errorf("failed to create scanner: %w", err)
        }
        p.scanner = scanner

        // Create writer with options
        p.writer = NewWriter(opts.DryRun, opts.Backup)

        // Scan for files
        scanResult, err := scanner.Scan(ctx)
        if err != nil {
                return nil, fmt.Errorf("failed to scan files: %w", err)
        }

        // Process each file
        for _, file := range scanResult.Files {
                // Check for context cancellation
                select {
                case <-ctx.Done():
                        return result, ctx.Err()
                default:
                }

                fileResult := p.processFile(ctx, file, mode)
                result.FileResults = append(result.FileResults, *fileResult)

                // Update counters
                result.TotalFiles++
                switch fileResult.Status {
                case StatusHasCopyright:
                        result.FilesWithCopyright++
                case StatusMissingCopyright:
                        result.FilesMissingCopyright++
                case StatusFixed:
                        result.FilesFixed++
                case StatusSkipped:
                        result.FilesSkipped++
                case StatusFailed:
                        result.FilesFailed++
                }
        }

        // Handle LICENSE file
        if mode == ModeFix {
                licensePath, licenseExists := p.license.FindLicense(absPath)
                if licenseExists {
                        result.LicenseStatus = fmt.Sprintf("LICENSE exists at %s", licensePath)
                } else {
                        // Try to create if configured
                        if p.config.License.AutoCreate {
                                if err := p.license.Create(absPath); err != nil {
                                        result.LicenseStatus = fmt.Sprintf("Failed to create LICENSE: %v", err)
                                } else {
                                        result.LicenseCreated = true
                                        result.LicenseStatus = "LICENSE created"
                                }
                        } else {
                                result.LicenseStatus = "No LICENSE file found (auto-create disabled)"
                        }
                }
        }

        return result, nil
}

// processFile handles processing of a single file
func (p *Processor) processFile(ctx context.Context, file files.FileInfo, mode Mode) *FileResult {
        result := &FileResult{
                FilePath:     file.Path,
                RelativePath: file.RelativePath,
        }

        // Skip binary files
        if file.IsBinary {
                result.Status = StatusSkipped
                result.Message = "binary file"
                return result
        }

        // Get language handler
        lang := p.registry.GetLanguage(file.Filename)
        if lang == nil {
                result.Status = StatusSkipped
                result.Message = "unsupported file type"
                return result
        }

        // Skip files that don't support comments
        if !lang.CanHaveComments() {
                result.Status = StatusSkipped
                result.Message = "file type does not support comments"
                return result
        }

        // Check if file has copyright
        checkResult, err := p.checker.Check(ctx, file.Path)
        if err != nil {
                result.Status = StatusFailed
                result.Error = err
                result.Message = fmt.Sprintf("failed to check file: %v", err)
                return result
        }

        if checkResult.Error != nil {
                result.Status = StatusFailed
                result.Error = checkResult.Error
                result.Message = fmt.Sprintf("check error: %v", checkResult.Error)
                return result
        }

        // If file has copyright, we're done
        if checkResult.HasCopyright {
                result.Status = StatusHasCopyright
                result.Message = fmt.Sprintf("copyright found at line %d", checkResult.LineFound)
                return result
        }

        // File is missing copyright
        if mode == ModeCheck {
                result.Status = StatusMissingCopyright
                result.Message = "copyright header missing"
                return result
        }

        // Mode is ModeFix - add the copyright header
        if err := p.addCopyrightHeader(file.Path, lang); err != nil {
                result.Status = StatusFailed
                result.Error = err
                result.Message = fmt.Sprintf("failed to add copyright: %v", err)
                return result
        }

        result.Status = StatusFixed
        result.Message = "copyright header added"
        return result
}

// addCopyrightHeader adds a copyright header to a file
func (p *Processor) addCopyrightHeader(filePath string, lang language.Language) error {
        // Generate the copyright text
        copyrightText, err := p.generator.Generate()
        if err != nil {
                return fmt.Errorf("failed to generate copyright: %w", err)
        }

        // Format the header for the specific language
        header := lang.FormatHeader(copyrightText)

        // Write the header to the file
        if err := p.writer.WriteHeader(filePath, header, lang); err != nil {
                return fmt.Errorf("failed to write header: %w", err)
        }

        return nil
}

// Writer handles writing copyright headers to files
type Writer struct {
        dryRun bool
        backup bool
}

// NewWriter creates a new writer with the specified options
func NewWriter(dryRun, backup bool) *Writer {
        return &Writer{
                dryRun: dryRun,
                backup: backup,
        }
}

// WriteHeader inserts a copyright header into a file
// It uses the language's PreserveLines and InsertPosition to correctly
// place the header after shebangs/encoding declarations
func (w *Writer) WriteHeader(filePath string, header string, lang language.Language) error {
        // Read the current file content
        content, err := os.ReadFile(filePath)
        if err != nil {
                return fmt.Errorf("failed to read file: %w", err)
        }

        // Get lines that should be preserved (shebang, encoding, etc.)
        preservedLines := lang.PreserveLines(content)

        // Build the new content
        var newContent strings.Builder

        // Write preserved lines
        for _, line := range preservedLines {
                newContent.WriteString(line)
        }

        // Add a blank line after preserved content if there are preserved lines
        if len(preservedLines) > 0 {
                // Check if the last preserved line already ends with newline
                lastLine := preservedLines[len(preservedLines)-1]
                if !strings.HasSuffix(lastLine, "\n") {
                        newContent.WriteString("\n")
                }
                newContent.WriteString("\n")
        }

        // Write the copyright header
        newContent.WriteString(header)

        // Add a blank line after the header
        if !strings.HasSuffix(header, "\n\n") {
                if strings.HasSuffix(header, "\n") {
                        newContent.WriteString("\n")
                } else {
                        newContent.WriteString("\n\n")
                }
        }

        // Get the insert position (byte offset after preserved lines)
        insertPos := lang.InsertPosition(content)

        // Write the remaining content after the insert position
        if insertPos < len(content) {
                remainingContent := content[insertPos:]
                // Skip leading blank lines in remaining content to avoid too much whitespace
                remainingStr := strings.TrimLeft(string(remainingContent), "\n")
                newContent.WriteString(remainingStr)
        }

        // Ensure file ends with a newline
        finalContent := newContent.String()
        if !strings.HasSuffix(finalContent, "\n") {
                finalContent += "\n"
        }

        // Write the new content to the file
        return w.WriteFile(filePath, []byte(finalContent))
}

// WriteFile writes content to a file (with backup if enabled)
func (w *Writer) WriteFile(filePath string, content []byte) error {
        // If dry-run mode, don't actually write
        if w.dryRun {
                return nil
        }

        // Create backup if enabled
        if w.backup {
                backupPath := filePath + ".bak"
                originalContent, err := os.ReadFile(filePath)
                if err != nil {
                        return fmt.Errorf("failed to read original file for backup: %w", err)
                }
                if err := os.WriteFile(backupPath, originalContent, 0644); err != nil {
                        return fmt.Errorf("failed to create backup file: %w", err)
                }
        }

        // Write the new content
        if err := os.WriteFile(filePath, content, 0644); err != nil {
                return fmt.Errorf("failed to write file: %w", err)
        }

        return nil
}

                fmt.Println()

        fmt.Println(result.Summary())

        if result.LicenseStatus != "" {
                fmt.Printf("License: %s\n", result.LicenseStatus)
        }
